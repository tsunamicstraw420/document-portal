Option 1: Official X API (Recommended for Production)

Pros:

  • Official support and reliability
  • Access to all X platform data
  • Authentication via Bearer Token (OAuth 2.0)
  • Rate limits documented and manageable

Cons:

  • Requires API approval and credentials
  • Cost for high-volume access
  • Setup requires Bearer Token

from playwright.async_api import async_playwright

async def scrape_tweet():
    async with async_playwright() as p:
        browser = await p.chromium.launch()
        page = await browser.new_page()
        await page.goto("https://x.com/cyrilXBT/status/2052235121416188114")
        content = await page.content()
        await browser.close()

Step 2: Extract Tweet Data

import requests
import json

def get_tweet(tweet_id, bearer_token):
    """Fetch tweet data from X API v2"""
    url = f"https://api.twitter.com/2/tweets/{tweet_id}"
    
    params = {
        "tweet.fields": "created_at,public_metrics,author_id",
        "expansions": "author_id",
        "user.fields": "username,verified"
    }
    
    headers = {"Authorization": f"Bearer {bearer_token}"}
    
    response = requests.get(url, headers=headers, params=params)
    return response.json()

# Usage
tweet_data = get_tweet("2052235121416188114", bearer_token)

Step 3: Parse and Analyze Content

def analyze_tweet(tweet_data):
    """Extract key information from tweet"""
    tweet = tweet_data['data']
    includes = tweet_data.get('includes', {})
    
    analysis = {
        'id': tweet['id'],
        'text': tweet['text'],
        'created_at': tweet['created_at'],
        'likes': tweet['public_metrics']['like_count'],
        'retweets': tweet['public_metrics']['retweet_count'],
        'replies': tweet['public_metrics']['reply_count'],
        'author': includes['users'][0]['username'] if includes.get('users') else None
    }
    
    return analysis

Step 4: Create Automation (Python Scheduler)

from schedule import Scheduler
from datetime import datetime

def scheduled_extraction():
    """Run tweet extraction on schedule"""
    tweet_data = get_tweet("2052235121416188114", bearer_token)
    analysis = analyze_tweet(tweet_data)
    
    # Save to file or database
    with open(f"tweets_{datetime.now().date()}.json", "w") as f:
        json.dump(analysis, f, indent=2)

# Schedule daily at 9 AM
scheduler = Scheduler()
scheduler.every().day.at("09:00").do(scheduled_extraction)

Part 5: Security Best Practices

✅ DO:

  • Store Bearer tokens in .env files (git-ignored)
  • Use environment variables for sensitive data
  • Rotate tokens regularly
  • Log API calls for audit trails

❌ DON'T:

  • Commit API keys to GitHub
  • Hardcode credentials in code
  • Share bearer tokens in chat/email
  • Use expired/revoked tokens
