"""
bot.py — Polymarket Latency Arbitrage Bot (Paper Mode Optimized)
"""
import argparse
import signal
import sys
import time
import threading

# ── Local imports ─────────────────────
import config
from logger_setup import log
from notifier import notifier
from price_feed import price_feed
from market_scanner import MarketScanner
from strategy import Strategy
from risk_manager import RiskManager
from order_manager import OrderManager

# ── Paper-mode virtual portfolio ──────────────────────────────────────────────
PAPER_STARTING_BALANCE = 1000.0

class ArbBot:
    def __init__(self, paper_mode: bool = True, scan_interval: float = config.SCAN_INTERVAL_SEC):
        self.paper_mode = paper_mode
        self.scan_interval = scan_interval
        self._running = False
        self._shutdown_event = threading.Event()

        # Load credentials and client
        creds_env = config.Credentials.from_env()
        self._creds_env = creds_env

        # Use fixed client from config.py
        self._clob = config.get_clob_client()

        # Determine starting balance
        starting_balance = self._resolve_starting_balance()

        # Compose subsystems
        self._risk = RiskManager(starting_balance=starting_balance)
        self._scanner = MarketScanner(clob_client=self._clob)
        self._strategy = Strategy(price_feed=price_feed, market_scanner=self._scanner)
        self._orders = OrderManager(
            clob_client=self._clob,
            risk_manager=self._risk,
            paper_mode=paper_mode,
        )

        # Cooldown tracking
        self._last_trade_time: dict[str, float] = {}
        self._trade_cooldown_sec = 30.0

        # Status timing
        self._last_status_time = 0.0

        mode = "PAPER" if paper_mode else "LIVE"
        log.info("=" * 60)
        log.info("Polymarket Latency Arbitrage Bot — %s MODE", mode)
        log.info("Starting balance: %.2f USDC", starting_balance)
        log.info("Edge threshold: %.1f%% (testing)", config.EDGE_THRESHOLD * 100)
        log.info("Scan interval: %.1fs", scan_interval)
        log.info("=" * 60)

    # ── Startup ───────────────────────────────────────────────────────────────
    def start(self) -> None:
        self._running = True
        price_feed.start()

        log.info("Waiting for Binance price feed to warm up...")
        deadline = time.monotonic() + 15.0
        while not price_feed.all_ready():
            if time.monotonic() > deadline:
                log.warning("Price feed warm-up timed out — continuing anyway")
                break
            time.sleep(0.5)
        log.info("Price feed ready.")

        self._scanner.start()

        signal.signal(signal.SIGINT, self._handle_signal)
        signal.signal(signal.SIGTERM, self._handle_signal)

        notifier.send(
            f"Bot started in {'PAPER' if self.paper_mode else 'LIVE'} mode. "
            f"Balance: {self._risk.current_balance:.2f} USDC"
        )

        self._run()

    def stop(self) -> None:
        log.info("Shutting down...")
        self._running = False
        price_feed.stop()
        self._scanner.stop()
        self._shutdown_event.set()
        log.info("Final state: %s", self._risk.summary())
        notifier.send(f"Bot stopped. Final state: {self._risk.summary()}")

    # ── Main loop ─────────────────────────────────────────────────────────────
    def _run(self) -> None:
        log.info("Main loop started (scan interval=%.1fs)", self.scan_interval)
        while self._running:
            loop_start = time.monotonic()
            try:
                self._iteration()
            except Exception as exc:
                log.error("Unhandled exception in main loop: %s", exc, exc_info=True)
                notifier.error(f"Main loop error: {exc}")

            elapsed = time.monotonic() - loop_start
            sleep_time = max(0.0, self.scan_interval - elapsed)
            self._shutdown_event.wait(timeout=sleep_time)
        log.info("Main loop exited.")

    def _iteration(self) -> None:
        """One scan cycle"""
        if self._risk.is_killed:
            log.critical("Kill switch active — shutting down.")
            self.stop()
            return

        if price_feed.any_stale():
            log.warning("Price feed is stale — skipping iteration")
            return

        # Print status every 15 seconds
        now = time.monotonic()
        if now - self._last_status_time > config.STATUS_PRINT_INTERVAL:
            self._log_status()
            self._last_status_time = now

        # Skip if risk limits prevent trading
        if not self._risk.trading_allowed:
            return

        # Evaluate signals
        signals = self._strategy.scan(bankroll=self._risk.current_balance)
        if not signals:
            return

        # Execute best signal
        for signal in signals:
            token_id = signal.token_id
            last = self._last_trade_time.get(token_id, 0)
            if time.time() - last < self._trade_cooldown_sec:
                continue

            result = self._orders.execute_signal(signal)
            if result and getattr(result, 'success', False):
                self._last_trade_time[token_id] = time.time()
                break

    # ── Helpers ───────────────────────────────────────────────────────────────
    def _resolve_starting_balance(self) -> float:
        if self.paper_mode:
            return PAPER_STARTING_BALANCE
        # Live balance logic can be added later
        return PAPER_STARTING_BALANCE

    def _log_status(self) -> None:
        """Clean status log every 15 seconds"""
        active_markets = len(self._scanner.get_markets())
        log.info(
            "STATUS | Balance=%.2f USDC | Active Markets=%d | Edge=%.1f%% | OpenPos=0",
            self._risk.current_balance,
            active_markets,
            config.EDGE_THRESHOLD * 100
        )

        # Price feed info using the global price_feed
        for sym in config.BINANCE_SYMBOLS:
            price = price_feed.latest_price(sym) if hasattr(price_feed, 'latest_price') else None
            change = price_feed.price_change_pct(sym, lookback_sec=config.LOOKBACK_SECONDS) * 100 if hasattr(price_feed, 'price_change_pct') else 0
            log.info("  %s price=%.2f  Δ%ds=%+.3f%%", sym.upper(), price or 0, config.LOOKBACK_SECONDS, change)

    def _handle_signal(self, signum, frame) -> None:
        log.info("Signal %s received — shutting down", signum)
        self.stop()


# ── Entry point ───────────────────────────────────────────────────────────────
def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser(description="Polymarket Latency Arbitrage Bot")
    parser.add_argument("--live", action="store_true", default=False)
    parser.add_argument("--interval", type=float, default=config.SCAN_INTERVAL_SEC)
    parser.add_argument("--balance", type=float, default=None)
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    if args.balance is not None:
        global PAPER_STARTING_BALANCE
        PAPER_STARTING_BALANCE = args.balance

    paper_mode = not args.live

    if not paper_mode:
        print("\n" + "!" * 60)
        print("! LIVE TRADING MODE — REAL USDC WILL BE SPENT")
        print("! Press Ctrl-C within 5 seconds to abort...")
        print("!" * 60 + "\n")
        try:
            time.sleep(5)
        except KeyboardInterrupt:
            sys.exit(0)

    bot = ArbBot(paper_mode=paper_mode, scan_interval=args.interval)
    bot.start()


if __name__ == "__main__":
    main()
